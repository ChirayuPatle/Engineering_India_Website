/**
 * ============================================
 * HACKATHON REGISTRATION - GOOGLE SHEETS SYNC
 * ============================================
 * 
 * This script receives hackathon registration data from your Next.js app
 * and automatically appends it to your Google Sheet.
 * 
 * SETUP INSTRUCTIONS:
 * 1. Create a Google Sheet named "Hackathon Registrations"
 * 2. Rename Sheet1 to "Registrations"
 * 3. Add headers in Row 1 (see below)
 * 4. Copy this entire script to Apps Script (Extensions → Apps Script)
 * 5. Deploy as Web App (Deploy → New deployment → Web app)
 * 6. Set "Who has access" to "Anyone"
 * 7. Copy the Web App URL to your .env file
 * 
 * HEADERS (Row 1):
 * Registration ID | Timestamp | Team Name | Leader Name | Leader Email | 
 * Leader Phone | Leader Gender | Institute | Branch | Year | Team Size | 
 * Member 1 Name | Member 1 Email | Member 1 Phone | Member 1 Gender | 
 * Member 1 Branch | Member 1 Year | Member 2 Name | Member 2 Email | 
 * Member 2 Phone | Member 2 Gender | Member 2 Branch | Member 2 Year | 
 * Member 3 Name | Member 3 Email | Member 3 Phone | Member 3 Gender | 
 * Member 3 Branch | Member 3 Year | Transaction ID | Payment Screenshot URL | Status
 */

/**
 * Handle POST requests from Next.js app
 * This function is called automatically when data is sent to the Web App URL
 */
function doPost(e) {
  try {
    // Log incoming request for debugging
    Logger.log('Received POST request');
    
    // Parse the JSON data sent from Next.js
    const data = JSON.parse(e.postData.contents);
    Logger.log('Parsed data: ' + JSON.stringify(data));
    
    // Get the active spreadsheet and "Registrations" sheet
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Registrations');
    
    if (!sheet) {
      throw new Error('Sheet "Registrations" not found. Please create it and add headers.');
    }
    
    // Extract team members (up to 3 additional members)
    const members = data.teamMembers || [];
    const member1 = members[0] || {};
    const member2 = members[1] || {};
    const member3 = members[2] || {};
    
    // Prepare row data matching the header columns
    const rowData = [
      data.registrationId || '',           // Registration ID
      data.timestamp || new Date(),        // Timestamp
      data.teamName || '',                 // Team Name
      data.teamLeaderName || '',           // Leader Name
      data.teamLeaderEmail || '',          // Leader Email
      data.teamLeaderPhone || '',          // Leader Phone
      data.teamLeaderGender || '',         // Leader Gender
      data.institute || '',                // Institute
      data.branch || '',                   // Branch
      data.year || '',                     // Year
      data.teamSize || 1,                  // Team Size
      
      // Member 1
      member1.name || '',
      member1.email || '',
      member1.phone || '',
      member1.gender || '',
      member1.branch || '',
      member1.year || '',
      
      // Member 2
      member2.name || '',
      member2.email || '',
      member2.phone || '',
      member2.gender || '',
      member2.branch || '',
      member2.year || '',
      
      // Member 3
      member3.name || '',
      member3.email || '',
      member3.phone || '',
      member3.gender || '',
      member3.branch || '',
      member3.year || '',
      
      // Payment & Status
      data.transactionId || '',            // Transaction ID
      data.paymentScreenshotUrl || '',     // Payment Screenshot URL
      data.status || 'pending'             // Status
    ];
    
    // Append the new row to the sheet
    sheet.appendRow(rowData);
    
    // Log success
    Logger.log('Successfully added registration: ' + data.teamName);
    
    // Return success response
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      message: 'Registration added to sheet successfully',
      teamName: data.teamName
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    // Log error for debugging
    Logger.log('Error: ' + error.toString());
    
    // Return error response
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * Handle GET requests (for testing)
 * Visit your Web App URL in browser to test if it's working
 */
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    message: 'Hackathon Registration Google Sheets API is running!',
    instructions: 'Send POST requests with registration data to sync with sheet.',
    timestamp: new Date().toISOString()
  })).setMimeType(ContentService.MimeType.JSON);
}

/**
 * OPTIONAL: Clear all data except headers
 * Run this manually to reset the sheet (CAUTION: Deletes all registrations!)
 */
function clearAllData() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Registrations');
  
  if (!sheet) {
    Logger.log('❌ ERROR: Sheet "Registrations" not found!');
    return;
  }
  
  const lastRow = sheet.getLastRow();
  
  if (lastRow > 1) {
    // Delete all rows except header (row 1)
    sheet.deleteRows(2, lastRow - 1);
    Logger.log('✅ Cleared ' + (lastRow - 1) + ' rows of data');
  } else {
    Logger.log('ℹ️ No data to clear (only headers present)');
  }
}

/**
 * OPTIONAL: Setup function to create headers automatically
 * Run this manually if you haven't added headers yet
 */
function setupHeaders() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Registrations');
  
  if (!sheet) {
    Logger.log('❌ ERROR: Sheet "Registrations" not found!');
    Logger.log('Please create a sheet named "Registrations" first.');
    return;
  }
  
  // Check if headers already exist
  if (sheet.getLastRow() > 0) {
    const ui = SpreadsheetApp.getUi();
    const response = ui.alert(
      'Headers Already Exist',
      'The sheet already has data. Do you want to overwrite row 1 with new headers?',
      ui.ButtonSet.YES_NO
    );
    
    if (response !== ui.Button.YES) {
      Logger.log('ℹ️ Setup cancelled by user');
      return;
    }
  }
  
  // Define headers
  const headers = [
    'Registration ID',
    'Timestamp',
    'Team Name',
    'Leader Name',
    'Leader Email',
    'Leader Phone',
    'Leader Gender',
    'Institute',
    'Branch',
    'Year',
    'Team Size',
    'Member 1 Name',
    'Member 1 Email',
    'Member 1 Phone',
    'Member 1 Gender',
    'Member 1 Branch',
    'Member 1 Year',
    'Member 2 Name',
    'Member 2 Email',
    'Member 2 Phone',
    'Member 2 Gender',
    'Member 2 Branch',
    'Member 2 Year',
    'Member 3 Name',
    'Member 3 Email',
    'Member 3 Phone',
    'Member 3 Gender',
    'Member 3 Branch',
    'Member 3 Year',
    'Transaction ID',
    'Payment Screenshot URL',
    'Status'
  ];
  
  // Add headers to row 1
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  
  // Format headers
  sheet.getRange(1, 1, 1, headers.length)
    .setFontWeight('bold')
    .setBackground('#4285f4')
    .setFontColor('#ffffff')
    .setHorizontalAlignment('center');
  
  // Auto-resize columns
  for (let i = 1; i <= headers.length; i++) {
    sheet.autoResizeColumn(i);
  }
  
  // Freeze header row
  sheet.setFrozenRows(1);
  
  Logger.log('✅ Headers setup complete!');
  Logger.log('Total columns: ' + headers.length);
}

/**
 * OPTIONAL: Format the entire sheet for better readability
 * Run this manually to apply formatting
 */
function formatSheet() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Registrations');
  
  if (!sheet) {
    Logger.log('❌ ERROR: Sheet "Registrations" not found!');
    return;
  }
  
  // Freeze header row
  sheet.setFrozenRows(1);
  
  // Format headers
  const headerRange = sheet.getRange(1, 1, 1, sheet.getLastColumn());
  headerRange
    .setFontWeight('bold')
    .setBackground('#4285f4')
    .setFontColor('#ffffff')
    .setHorizontalAlignment('center');
  
  // Alternate row colors for data rows
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    const dataRange = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn());
    dataRange.applyRowBanding(SpreadsheetApp.BandingTheme.LIGHT_GREY, false, false);
  }
  
  // Auto-resize all columns
  for (let i = 1; i <= sheet.getLastColumn(); i++) {
    sheet.autoResizeColumn(i);
  }
  
  // Set column widths for better display
  sheet.setColumnWidth(1, 150);  // Registration ID
  sheet.setColumnWidth(2, 180);  // Timestamp
  sheet.setColumnWidth(3, 200);  // Team Name
  
  Logger.log('✅ Sheet formatting applied!');
}
