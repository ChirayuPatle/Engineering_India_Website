/**
 * Handle POST requests from Next.js app (Real-time sync)
 * Called automatically when a user submits a registration
 */
function doPost(e) {
  try {
    Logger.log('Received POST request');
    Logger.log('Request object: ' + JSON.stringify(e));
    
    // Check if postData exists
    if (!e || !e.postData || !e.postData.contents) {
      throw new Error('No data received in POST request. Make sure you are sending JSON data in the request body.');
    }
    
    // Parse the JSON data sent from Next.js
    const data = JSON.parse(e.postData.contents);
    Logger.log('Parsed data: ' + JSON.stringify(data));
    
    // Get the sheet
    const sheet = getOrCreateSheet();
    
    // Add the registration
    addRegistrationToSheet(sheet, data);
    
    Logger.log('Successfully added registration: ' + data.teamName);
    
    // Return success response
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      message: 'Registration added to sheet successfully',
      teamName: data.teamName
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    Logger.log('Error in doPost: ' + error.toString());
    
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * Handle GET requests (for testing)
 */
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    message: 'Hackathon Registration Google Sheets API is running!',
    instructions: 'Send POST requests with registration data to sync with sheet.',
    timestamp: new Date().toISOString()
  })).setMimeType(ContentService.MimeType.JSON);
}

/**
 * Main sync function - Fetches all registrations from database and syncs to sheet
 * Run this manually or set up as a time-triggered function
 */
function syncFromDatabase() {
  const scriptProperties = PropertiesService.getScriptProperties();
  const vercelAppUrl = scriptProperties.getProperty('VERCEL_APP_URL');
  const apiKey = scriptProperties.getProperty('GOOGLE_APP_SCRIPT_API_KEY');

  if (!vercelAppUrl || !apiKey) {
    Logger.log('❌ ERROR: VERCEL_APP_URL or GOOGLE_APP_SCRIPT_API_KEY not set in Script Properties.');
    Logger.log('Please go to Project Settings (⚙️) and add them under Script Properties.');
    return;
  }

  try {
    Logger.log('🔄 Starting database sync...');
    
    // Fetch all registrations from database
    const data = fetchRegistrationsFromDatabase(vercelAppUrl, apiKey);
    
    if (!data || data.length === 0) {
      Logger.log('ℹ️ No registrations found in database.');
      return;
    }
    
    Logger.log(`📊 Found ${data.length} registrations in database`);
    
    // Get or create sheet
    const sheet = getOrCreateSheet();
    
    // Sync data to sheet
    syncDataToSheet(sheet, data);
    
    Logger.log('✅ Database sync completed successfully!');
    
  } catch (e) {
    Logger.log('❌ Error during sync: ' + e.toString());
  }
}

/**
 * Fetch all registrations from the Next.js API
 */
function fetchRegistrationsFromDatabase(appUrl, apiKey) {
  const endpoint = appUrl + '/api/hackathon/sync';

  const options = {
    'method': 'get',
    'headers': {
      'x-api-key': apiKey,
      'Content-Type': 'application/json'
    },
    'muteHttpExceptions': true
  };

  Logger.log('Sending request to: ' + endpoint);

  const response = UrlFetchApp.fetch(endpoint, options);
  const responseCode = response.getResponseCode();
  const responseBody = response.getContentText();

  Logger.log('API Response Code: ' + responseCode);

  if (responseCode === 200) {
    const jsonResponse = JSON.parse(responseBody);
    if (jsonResponse && jsonResponse.success && jsonResponse.data) {
      return jsonResponse.data;
    }
    return [];
  } else {
    Logger.log('API Response Body: ' + responseBody);
    throw new Error('Failed to fetch data from API. Response: ' + responseBody);
  }
}

/**
 * Sync data to sheet, handling duplicates and updates
 */
function syncDataToSheet(sheet, newData) {
  // Get existing data from the sheet
  const existingData = sheet.getDataRange().getValues();
  const existingHeaders = existingData.length > 0 ? existingData[0] : [];
  const existingRecords = existingData.slice(1); // Skip headers

  const newRecordsToAdd = [];
  const updatedRecords = {}; // Map to store records that need updating

  // Get headers from new data
  const newHeaders = Object.keys(newData[0]);

  // Map existing records by their ID for efficient lookup
  const existingRecordsMap = new Map();
  let idColumnIndex = -1;
  
  if (existingHeaders.includes('id')) {
    idColumnIndex = existingHeaders.indexOf('id');
    existingRecords.forEach(row => {
      if (row[idColumnIndex]) {
        existingRecordsMap.set(row[idColumnIndex].toString(), row);
      }
    });
  } else {
    Logger.log('⚠️ Warning: "id" column not found. Will add all records as new.');
  }

  // Check each record from database
  newData.forEach(newRecord => {
    const recordId = newRecord.id ? newRecord.id.toString() : null;

    if (recordId && existingRecordsMap.has(recordId)) {
      // Record exists - check if it needs updating
      const existingRow = existingRecordsMap.get(recordId);
      let needsUpdate = false;
      const updatedRow = [];

      for (let i = 0; i < newHeaders.length; i++) {
        const header = newHeaders[i];
        const newValue = newRecord[header];
        const existingColIndex = existingHeaders.indexOf(header);
        const existingValue = existingColIndex >= 0 ? existingRow[existingColIndex] : null;

        if (String(newValue) !== String(existingValue)) {
          needsUpdate = true;
        }
        updatedRow.push(newValue);
      }

      if (needsUpdate) {
        updatedRecords[recordId] = updatedRow;
        Logger.log(`📝 Will update record: ${newRecord.teamName}`);
      }
    } else {
      // New record - add to list
      const row = [];
      newHeaders.forEach(header => row.push(newRecord[header]));
      newRecordsToAdd.push(row);
      Logger.log(`➕ Will add new record: ${newRecord.teamName}`);
    }
  });

  // If sheet is empty or headers don't match, write headers
  if (existingData.length === 0 || JSON.stringify(existingHeaders) !== JSON.stringify(newHeaders)) {
    if (existingData.length > 0) {
      Logger.log('📋 Headers changed, updating...');
    }
    sheet.getRange(1, 1, 1, newHeaders.length).setValues([newHeaders]);
    formatHeaders(sheet, newHeaders.length);
  }

  // Append new records
  if (newRecordsToAdd.length > 0) {
    const startRow = sheet.getLastRow() + 1;
    sheet.getRange(startRow, 1, newRecordsToAdd.length, newHeaders.length).setValues(newRecordsToAdd);
    Logger.log(`✅ Added ${newRecordsToAdd.length} new records`);
  }

  // Update existing records
  if (Object.keys(updatedRecords).length > 0) {
    const allSheetData = sheet.getDataRange().getValues();
    const headers = allSheetData[0];
    const idColIndex = headers.indexOf('id');

    if (idColIndex !== -1) {
      for (let r = 1; r < allSheetData.length; r++) {
        const rowId = allSheetData[r][idColIndex] ? allSheetData[r][idColIndex].toString() : null;
        if (rowId && updatedRecords[rowId]) {
          const updatedRow = updatedRecords[rowId];
          const rowToSet = [];
          
          for (let c = 0; c < headers.length; c++) {
            const headerName = headers[c];
            const newHeaderIndex = newHeaders.indexOf(headerName);
            if (newHeaderIndex !== -1) {
              rowToSet.push(updatedRow[newHeaderIndex]);
            } else {
              rowToSet.push(allSheetData[r][c]);
            }
          }
          
          sheet.getRange(r + 1, 1, 1, rowToSet.length).setValues([rowToSet]);
        }
      }
      Logger.log(`✅ Updated ${Object.keys(updatedRecords).length} existing records`);
    }
  }

  if (newRecordsToAdd.length === 0 && Object.keys(updatedRecords).length === 0) {
    Logger.log('✅ All data is already in sync!');
  }
}

/**
 * Get existing sheet or create if it doesn't exist
 */
function getOrCreateSheet() {
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = spreadsheet.getSheetByName(SHEET_NAME);
  
  if (!sheet) {
    sheet = spreadsheet.insertSheet(SHEET_NAME);
    Logger.log('Created new sheet: ' + SHEET_NAME);
    
    // Setup headers for new sheet
    setupHeaders();
    sheet = spreadsheet.getSheetByName(SHEET_NAME); // Re-get after setup
  }
  
  return sheet;
}

/**
 * Add a single registration to the sheet
 */
function addRegistrationToSheet(sheet, data) {
  // Prepare row data
  const rowData = [
    data.id || data.registrationId || '',
    data.registrationId || '',
    data.timestamp || new Date(),
    data.teamName || '',
    data.leaderName || data.teamLeaderName || '',
    data.leaderEmail || data.teamLeaderEmail || '',
    data.leaderPhone || data.teamLeaderPhone || '',
    data.leaderGender || data.teamLeaderGender || '',
    data.institute || '',
    data.branch || '',
    data.year || '',
    data.teamSize || 1,
    
    // Member 1
    data.member1Name || '',
    data.member1Email || '',
    data.member1Phone || '',
    data.member1Gender || '',
    data.member1Branch || '',
    data.member1Year || '',
    
    // Member 2
    data.member2Name || '',
    data.member2Email || '',
    data.member2Phone || '',
    data.member2Gender || '',
    data.member2Branch || '',
    data.member2Year || '',
    
    // Member 3
    data.member3Name || '',
    data.member3Email || '',
    data.member3Phone || '',
    data.member3Gender || '',
    data.member3Branch || '',
    data.member3Year || '',
    
    // Payment
    data.transactionId || 'N/A',
    data.paymentScreenshotUrl || '',
    data.status || 'pending'
  ];
  
  // Check if this registration already exists (by ID)
  const existingData = sheet.getDataRange().getValues();
  if (existingData.length > 1) {
    const headers = existingData[0];
    const idIndex = headers.indexOf('id');
    
    if (idIndex >= 0 && (data.id || data.registrationId)) {
      const searchId = (data.id || data.registrationId).toString();
      
      for (let i = 1; i < existingData.length; i++) {
        if (existingData[i][idIndex]?.toString() === searchId) {
          Logger.log('Registration already exists, skipping: ' + data.teamName);
          return;
        }
      }
    }
  }
  
  // Append the row
  sheet.appendRow(rowData);
}

/**
 * Format headers with styling
 */
function formatHeaders(sheet, numColumns) {
  const headerRange = sheet.getRange(1, 1, 1, numColumns);
  headerRange
    .setFontWeight('bold')
    .setBackground('#4285f4')
    .setFontColor('#ffffff')
    .setHorizontalAlignment('center');
  
  sheet.setFrozenRows(1);
}
