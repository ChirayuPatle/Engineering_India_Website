/**
 * Handle POST requests from Next.js app (Real-time sync)
 * Called automatically when a user submits a registration
 */
function doPost(e) {
  try {
    Logger.log('Received POST request');
    Logger.log('Request object: ' + JSON.stringify(e));
    
    // Check if postData exists
    if (!e || !e.postData || !e.postData.contents) {
      throw new Error('No data received in POST request. Make sure you are sending JSON data in the request body.');
    }
    
    // Parse the JSON data sent from Next.js
    const data = JSON.parse(e.postData.contents);
    Logger.log('Parsed data: ' + JSON.stringify(data));
    
    // Get the sheet
    const sheet = getOrCreateSheet();
    
    // Add the registration
    addRegistrationToSheet(sheet, data);
    
    Logger.log('Successfully added registration: ' + data.teamName);
    
    // Return success response
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      message: 'Registration added to sheet successfully',
      teamName: data.teamName
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    Logger.log('Error in doPost: ' + error.toString());
    
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * Handle GET requests (for testing)
 */
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    message: 'Hackathon Registration Google Sheets API is running!',
    instructions: 'Send POST requests with registration data to sync with sheet.',
    timestamp: new Date().toISOString()
  })).setMimeType(ContentService.MimeType.JSON);
}

/**
 * Main sync function - Fetches all registrations from database and syncs to sheet
 * Run this manually or set up as a time-triggered function
 */
function syncFromDatabase() {
  const scriptProperties = PropertiesService.getScriptProperties();
  const vercelAppUrl = scriptProperties.getProperty('VERCEL_APP_URL');
  const apiKey = scriptProperties.getProperty('GOOGLE_APP_SCRIPT_API_KEY');

  if (!vercelAppUrl || !apiKey) {
    Logger.log('❌ ERROR: VERCEL_APP_URL or GOOGLE_APP_SCRIPT_API_KEY not set in Script Properties.');
    Logger.log('Please go to Project Settings (⚙️) and add them under Script Properties.');
    return;
  }

  try {
    Logger.log('🔄 Starting database sync...');
    
    // Fetch all registrations from database
    const data = fetchRegistrationsFromDatabase(vercelAppUrl, apiKey);
    
    if (!data || data.length === 0) {
      Logger.log('ℹ️ No registrations found in database.');
      return;
    }
    
    Logger.log(`📊 Found ${data.length} registrations in database`);
    
    // Get or create sheet
    const sheet = getOrCreateSheet();
    
    // Sync data to sheet
    syncDataToSheet(sheet, data);
    
    Logger.log('✅ Database sync completed successfully!');
    
  } catch (e) {
    Logger.log('❌ Error during sync: ' + e.toString());
  }
}

/**
 * Fetch all registrations from the Next.js API
 */
function fetchRegistrationsFromDatabase(appUrl, apiKey) {
  const endpoint = appUrl + '/api/hackathon/sync';

  const options = {
    'method': 'get',
    'headers': {
      'x-api-key': apiKey,
      'Content-Type': 'application/json'
    },
    'muteHttpExceptions': true
  };

  Logger.log('Sending request to: ' + endpoint);

  const response = UrlFetchApp.fetch(endpoint, options);
  const responseCode = response.getResponseCode();
  const responseBody = response.getContentText();

  Logger.log('API Response Code: ' + responseCode);

  if (responseCode === 200) {
    const jsonResponse = JSON.parse(responseBody);
    if (jsonResponse && jsonResponse.success && jsonResponse.data) {
      return jsonResponse.data;
    }
    return [];
  } else {
    Logger.log('API Response Body: ' + responseBody);
    throw new Error('Failed to fetch data from API. Response: ' + responseBody);
  }
}

/**
 * Sync data to sheet, handling duplicates and updates
 * Uses header-based mapping so column order doesn't matter
 */
function syncDataToSheet(sheet, newData) {
  // Get existing data from the sheet
  const existingData = sheet.getDataRange().getValues();
  const existingHeaders = existingData.length > 0 ? existingData[0] : [];
  const existingRecords = existingData.slice(1); // Skip headers

  const newRecordsToAdd = [];
  const recordsToUpdate = new Map(); // Map of row index to new data

  // Get the expected headers (standardized field names)
  const standardHeaders = [
    'id',
    'Registration ID',
    'Timestamp',
    'Team Name',
    'Leader Name',
    'Leader Email',
    'Leader Phone',
    'Leader Gender',
    'Institute',
    'Branch',
    'Year',
    'Team Size',
    'Member 1 Name',
    'Member 1 Email',
    'Member 1 Phone',
    'Member 1 Gender',
    'Member 1 Branch',
    'Member 1 Year',
    'Member 2 Name',
    'Member 2 Email',
    'Member 2 Phone',
    'Member 2 Gender',
    'Member 2 Branch',
    'Member 2 Year',
    'Member 3 Name',
    'Member 3 Email',
    'Member 3 Phone',
    'Member 3 Gender',
    'Member 3 Branch',
    'Member 3 Year',
    'Transaction ID',
    'Payment Screenshot URL',
    'Status'
  ];

  // Create header index map for existing data
  const headerIndexMap = {};
  existingHeaders.forEach((header, index) => {
    headerIndexMap[header] = index;
  });

  // Map existing records by their ID for efficient lookup
  const existingRecordsMap = new Map();
  const idColumnIndex = headerIndexMap['id'];
  
  if (idColumnIndex !== undefined) {
    existingRecords.forEach((row, rowIndex) => {
      if (row[idColumnIndex]) {
        existingRecordsMap.set(row[idColumnIndex].toString(), rowIndex + 1); // +1 because of header row
      }
    });
  } else {
    Logger.log('⚠️ Warning: "id" column not found. Will add all records as new.');
  }

  // Process each record from database
  newData.forEach(newRecord => {
    const recordId = newRecord.id ? newRecord.id.toString() : null;
    
    // Create data map for this record
    const dataMap = {
      'id': newRecord.id || newRecord.registrationId || '',
      'Registration ID': newRecord.registrationId || '',
      'Timestamp': newRecord.timestamp || new Date(),
      'Team Name': newRecord.teamName || '',
      'Leader Name': newRecord.leaderName || '',
      'Leader Email': newRecord.leaderEmail || '',
      'Leader Phone': newRecord.leaderPhone || '',
      'Leader Gender': newRecord.leaderGender || '',
      'Institute': newRecord.institute || '',
      'Branch': newRecord.branch || '',
      'Year': newRecord.year || '',
      'Team Size': newRecord.teamSize || 1,
      'Member 1 Name': newRecord.member1Name || '',
      'Member 1 Email': newRecord.member1Email || '',
      'Member 1 Phone': newRecord.member1Phone || '',
      'Member 1 Gender': newRecord.member1Gender || '',
      'Member 1 Branch': newRecord.member1Branch || '',
      'Member 1 Year': newRecord.member1Year || '',
      'Member 2 Name': newRecord.member2Name || '',
      'Member 2 Email': newRecord.member2Email || '',
      'Member 2 Phone': newRecord.member2Phone || '',
      'Member 2 Gender': newRecord.member2Gender || '',
      'Member 2 Branch': newRecord.member2Branch || '',
      'Member 2 Year': newRecord.member2Year || '',
      'Member 3 Name': newRecord.member3Name || '',
      'Member 3 Email': newRecord.member3Email || '',
      'Member 3 Phone': newRecord.member3Phone || '',
      'Member 3 Gender': newRecord.member3Gender || '',
      'Member 3 Branch': newRecord.member3Branch || '',
      'Member 3 Year': newRecord.member3Year || '',
      'Transaction ID': newRecord.transactionId || 'N/A',
      'Payment Screenshot URL': newRecord.paymentScreenshotUrl || '',
      'Status': newRecord.status || 'pending'
    };

    if (recordId && existingRecordsMap.has(recordId)) {
      // Record exists - check if it needs updating
      const existingRowIndex = existingRecordsMap.get(recordId);
      const existingRow = existingData[existingRowIndex];
      let needsUpdate = false;

      // Check each field to see if it changed
      for (let header of existingHeaders) {
        const existingValue = existingRow[headerIndexMap[header]];
        const newValue = dataMap[header];
        
        if (String(newValue) !== String(existingValue)) {
          needsUpdate = true;
          break;
        }
      }

      if (needsUpdate) {
        // Build row data based on existing header order
        const rowData = [];
        for (let header of existingHeaders) {
          rowData.push(dataMap[header] !== undefined ? dataMap[header] : '');
        }
        recordsToUpdate.set(existingRowIndex, rowData);
        Logger.log(`📝 Will update record: ${newRecord.teamName}`);
      }
    } else {
      // New record - add to list
      // Build row data based on existing header order (or standard if no headers)
      const headersToUse = existingHeaders.length > 0 ? existingHeaders : standardHeaders;
      const rowData = [];
      for (let header of headersToUse) {
        rowData.push(dataMap[header] !== undefined ? dataMap[header] : '');
      }
      newRecordsToAdd.push(rowData);
      Logger.log(`➕ Will add new record: ${newRecord.teamName}`);
    }
  });

  // If sheet is empty, add headers
  if (existingData.length === 0) {
    Logger.log('📋 Adding headers to new sheet...');
    sheet.getRange(1, 1, 1, standardHeaders.length).setValues([standardHeaders]);
    formatHeaders(sheet, standardHeaders.length);
  }

  // Append new records
  if (newRecordsToAdd.length > 0) {
    const startRow = sheet.getLastRow() + 1;
    const numCols = existingHeaders.length > 0 ? existingHeaders.length : standardHeaders.length;
    sheet.getRange(startRow, 1, newRecordsToAdd.length, numCols).setValues(newRecordsToAdd);
    Logger.log(`✅ Added ${newRecordsToAdd.length} new records`);
  }

  // Update existing records
  if (recordsToUpdate.size > 0) {
    recordsToUpdate.forEach((rowData, rowIndex) => {
      sheet.getRange(rowIndex + 1, 1, 1, rowData.length).setValues([rowData]);
    });
    Logger.log(`✅ Updated ${recordsToUpdate.size} existing records`);
  }

  if (newRecordsToAdd.length === 0 && recordsToUpdate.size === 0) {
    Logger.log('✅ All data is already in sync!');
  }
}

/**
 * Get existing sheet or create if it doesn't exist
 */
function getOrCreateSheet() {
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = spreadsheet.getSheetByName(SHEET_NAME);
  
  if (!sheet) {
    sheet = spreadsheet.insertSheet(SHEET_NAME);
    Logger.log('Created new sheet: ' + SHEET_NAME);
    
    // Setup headers for new sheet
    setupHeaders();
    sheet = spreadsheet.getSheetByName(SHEET_NAME); // Re-get after setup
  }
  
  return sheet;
}

/**
 * Add a single registration to the sheet using header-based mapping
 * This ensures data goes to the correct column regardless of column order
 */
function addRegistrationToSheet(sheet, data) {
  // Get headers from the sheet
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  
  // Check if this registration already exists (by ID)
  const existingData = sheet.getDataRange().getValues();
  if (existingData.length > 1) {
    const idIndex = headers.indexOf('id');
    
    if (idIndex >= 0 && (data.id || data.registrationId)) {
      const searchId = (data.id || data.registrationId).toString();
      
      for (let i = 1; i < existingData.length; i++) {
        if (existingData[i][idIndex]?.toString() === searchId) {
          Logger.log('Registration already exists, skipping: ' + data.teamName);
          return;
        }
      }
    }
  }
  
  // Create a mapping of field names to data values
  const dataMap = {
    'id': data.id || data.registrationId || '',
    'Registration ID': data.registrationId || '',
    'Timestamp': data.timestamp || new Date(),
    'Team Name': data.teamName || '',
    'Leader Name': data.leaderName || data.teamLeaderName || '',
    'Leader Email': data.leaderEmail || data.teamLeaderEmail || '',
    'Leader Phone': data.leaderPhone || data.teamLeaderPhone || '',
    'Leader Gender': data.leaderGender || data.teamLeaderGender || '',
    'Institute': data.institute || '',
    'Branch': data.branch || '',
    'Year': data.year || '',
    'Team Size': data.teamSize || 1,
    
    // Member 1
    'Member 1 Name': data.member1Name || '',
    'Member 1 Email': data.member1Email || '',
    'Member 1 Phone': data.member1Phone || '',
    'Member 1 Gender': data.member1Gender || '',
    'Member 1 Branch': data.member1Branch || '',
    'Member 1 Year': data.member1Year || '',
    
    // Member 2
    'Member 2 Name': data.member2Name || '',
    'Member 2 Email': data.member2Email || '',
    'Member 2 Phone': data.member2Phone || '',
    'Member 2 Gender': data.member2Gender || '',
    'Member 2 Branch': data.member2Branch || '',
    'Member 2 Year': data.member2Year || '',
    
    // Member 3
    'Member 3 Name': data.member3Name || '',
    'Member 3 Email': data.member3Email || '',
    'Member 3 Phone': data.member3Phone || '',
    'Member 3 Gender': data.member3Gender || '',
    'Member 3 Branch': data.member3Branch || '',
    'Member 3 Year': data.member3Year || '',
    
    // Payment
    'Transaction ID': data.transactionId || 'N/A',
    'Payment Screenshot URL': data.paymentScreenshotUrl || '',
    'Status': data.status || 'pending'
  };
  
  // Build row data based on actual header order
  const rowData = [];
  for (let i = 0; i < headers.length; i++) {
    const header = headers[i];
    rowData.push(dataMap[header] !== undefined ? dataMap[header] : '');
  }
  
  // Append the row
  sheet.appendRow(rowData);
  Logger.log('✅ Added registration: ' + data.teamName);
}

/**
 * Format headers with styling
 */
function formatHeaders(sheet, numColumns) {
  const headerRange = sheet.getRange(1, 1, 1, numColumns);
  headerRange
    .setFontWeight('bold')
    .setBackground('#4285f4')
    .setFontColor('#ffffff')
    .setHorizontalAlignment('center');
  
  sheet.setFrozenRows(1);
}
